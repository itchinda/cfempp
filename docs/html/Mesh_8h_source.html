195class LineMeshBuilder {
196public:
197 static void build(Mesh& mesh, int nx, const Vector3D& start, const Vector3D& end) {
198 mesh.clear();
199 for (int i = 0; i <= nx; ++i) {
200 double t = static_cast<double>(i) / nx;
201 Vector3D p = start * (1.0 - t) + end * t;
202 mesh.addVertex(p.x, p.y, p.z);
203 }
204
205 for (int i = 0; i < nx; ++i) {
206 mesh.addCell(CellType::LINE2, { (CfemInt)i, (CfemInt)i + 1 });
207 }
208
209 mesh.buildMesh();
210 if (mesh.getOrder() == 2) mesh.upgradeToQuadratic(); // Becomes LINE3
211 }

303class MeshQualityTool {
304public:
305 static QualityReport analyze(const Mesh& mesh) {
306 QualityReport report;
307
308 for (size_t i = 0; i < mesh.getNumCells(); ++i) {
309 const auto& cell = mesh.getCell(i);
310
311 // Signed Volume (Jacobian)
312 double vol = mesh.computeCellMeasure(cell.type, cell.vertex_ids);
313
314 if (vol < -cfem::ZERO_TOLERANCE) {
315 report.invertedCount++;
316 } else if (std::abs(vol) <= cfem::ZERO_TOLERANCE) {
317 report.degenerateCount++;
318 }
319
320 report.minVolume = std::min(report.minVolume, vol);
321 report.maxVolume = std::max(report.maxVolume, vol);
322 }
323
324 return report;
325 }
326
327 static void printSummary(const QualityReport& report) {
328 std::cout << "\n--- CFEM MESH QUALITY REPORT ---" << std::endl;
329 std::cout << "Min Volume: " << report.minVolume << std::endl;
330 std::cout << "Max Volume: " << report.maxVolume << std::endl;
331
332 if (report.invertedCount > 0) {
333 CFEM_WARNING("Mesh contains " + std::to_string(report.invertedCount) + " inverted elements!");
334 }
335 if (report.degenerateCount > 0) {
336 CFEM_WARNING("Mesh contains " + std::to_string(report.degenerateCount) + " degenerate (zero-volume) elements!");
337 }
338
339 if (report.invertedCount == 0 && report.degenerateCount == 0) {
340 std::cout << "Result: Mesh is valid for simulation." << std::endl;
341 }
342 std::cout << "--------------------------------\n" << std::endl;
343 }
