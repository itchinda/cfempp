    class LocalAssembler {

        <<interface>>
        +assemble(Cell, Context, Ke, Fe)*
    }

    class BoundaryAssembler {
        <<interface>>
        +assemble(Facet, Cell, Context, Ke, Fe)*
    }

    class InterfaceAssembler {
        <<interface>>
        +assemble(Facet, CellL, CellR, ContextL, ContextR, Ke_shared)*
    }

    class AssemblyEngine {
        -Mesh* mesh
        -FESpace* space
        -vector~CellAssembler*~ volumic_assemblers
        -vector~BoundaryAssembler*~ boundary_assemblers
        -vector~InterfaceAsembler*~ interface_assemblers
        +assemble_system()
        +assemble_cells()
        +assemble_boundary_facets()
        +assemble_interface_facets()
    }

    class PetscGlobalAssembler {

        +finalize()
    }
